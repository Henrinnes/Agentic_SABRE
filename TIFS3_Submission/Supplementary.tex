\section{Behavioural Feature Extraction Algorithm}\label{sec:supp_features}
% ─────────────────────────────────────────────────────────────────────────────

Algorithm~\ref{alg:features} describes the 16-dimensional feature extraction
procedure from the raw RanSMAP CSV telemetry files.
Each CSV row corresponds to a single I/O or memory event captured by the
BitVisor hypervisor~\cite{hirano2025ransmap}.
Features are computed over non-overlapping $T_{\text{window}}{=}10\,\text{s}$ windows.

\begin{algorithm}[h]
\caption{16-dimensional behavioural feature extraction}
\label{alg:features}
\begin{algorithmic}[1]
\Require CSV event stream $E = \{e_k\}_{k=1}^{n}$; window size
  $T_w{=}10\,\text{s}$
\Ensure Feature matrix $X \in \R^{W \times 16}$, one row per window

\State Partition $E$ into $W$ non-overlapping windows of duration $T_w$
\For{each window $w = 1, \ldots, W$}
  \State $E_w \gets \{e_k : t(e_k) \in [(w{-}1)T_w,\, wT_w)\}$
  \State \textit{// Storage features}
  \State $\bar{T}_{s,r} \gets \sum_{\text{read}} \text{bytes}/T_w$;
         $\bar{T}_{s,w} \gets \sum_{\text{write}} \text{bytes}/T_w$
  \State $V_{s,r} \gets \text{Var}(\{\text{LBA}(e) : e \in E_w,\,
         \text{type}(e){=}\text{read}\})$
  \State $V_{s,w} \gets \text{Var}(\{\text{LBA}(e) : e \in E_w,\,
         \text{type}(e){=}\text{write}\})$
  \State $\bar{H}_{s,w} \gets \text{mean}_{e \in E_w,\,\text{type}(e){=}\text{write}}
         H(\text{sector}(e))$
  \State $\chi_s \gets [\bar{T}_{s,r},\, \bar{T}_{s,w},\, V_{s,r},\,
         V_{s,w},\, \bar{H}_{s,w}]$ \Comment{5D storage}
  \State \textit{// Memory features}
  \State $H_{m,w} \gets H(\{\text{GPA}(e) : e \in E_w,\, o(e){=}w\})$
  \State $H_{m,rw} \gets H(\{\text{GPA}(e) : e \in E_w,\,
         o(e) \in \{r,w\}\})$
  \State For each $(s, o) \in \{4K,2M,\text{MMIO}\} \times \{r,w,rw,x\}$:
         compute $C_{s,o}$ as event count
  \State $V_{m,w} \gets \text{Var}(\{\text{GPA}(e) : e \in E_w,\,
         o(e){=}w\})$
  \State $\chi_m \gets [H_{m,w},\, H_{m,rw},\, C_{4K,r},\, C_{4K,w},\,$
  \Statex \hspace{3.2em}$C_{4K,rw},\, C_{4K,x},\, C_{2M,r},\, C_{2M,rw},$
  \Statex \hspace{3.2em}$C_{\text{MMIO},r},\, C_{\text{MMIO},w},\, V_{m,w}]$
         \Comment{11D}
  \State $X[w, :] \gets [\chi_s \,;\, \chi_m]$
         \hfill\Comment{16D feature row}
\EndFor
\State \Return $X$
\end{algorithmic}
\end{algorithm}

\section{Decision MLP Training Procedure}\label{sec:supp_mlp}
% ─────────────────────────────────────────────────────────────────────────────

Algorithm~\ref{alg:mlp_train} describes the full Decision MLP training
procedure including CTGAN augmentation, temperature calibration, and
Bayesian TPE policy optimisation.

\begin{algorithm}[h]
\caption{Decision MLP training with CTGAN augmentation and Bayesian TPE}
\label{alg:mlp_train}
\begin{algorithmic}[1]
\Require Trained agents $\{f_i\}_{i=1}^{3}$;
  labelled score-pool splits
  $\mathcal{D}_{\text{tr}}, \mathcal{D}_{\text{val}}, \mathcal{D}_{\text{cal}},
   \mathcal{D}_{\text{test}}$;
  $N_{\text{synth}}{=}2{,}000$, $\lambda{=}0.3$
  \Comment{$\mathcal{D}_{\text{cal}}$ is reused for both temperature
    scaling (Stage 4) and TPE (Stage 5) below, and is drawn independently
    of the session-level splits used for $\mathcal{D}_{\text{tr}}$/
    $\mathcal{D}_{\text{val}}$/$\mathcal{D}_{\text{test}}$ elsewhere in this
    paper -- see the corresponding limitation in the main paper's
    Discussion.}

\State \textit{// Stage 1: Score-pool construction}
\For{each split $D \in \{\mathcal{D}_{\text{val}},
    \mathcal{D}_{\text{cal}}, \mathcal{D}_{\text{test}}\}$}
  \State Compute $\mathbf{v}_j = (p_{\text{sem},j}, \sigma_{\text{sem},j},
    p_{\text{beh},j}, \sigma_{\text{beh},j}, p_{\text{know},j},
    \sigma_{\text{know},j})$ for each sample $j$ using $f_{\text{beh}}$
    trained on the full $\mathcal{D}_{\text{tr}}$
\EndFor
\State \textit{// $\mathcal{D}_{\text{tr}}$ requires out-of-fold scoring: partition
  $\mathcal{D}_{\text{tr}}$'s sessions into 5 session-grouped, stratified folds;
  for each fold, train $f_{\text{beh}}^{(k)}$ on the other four and score the
  held-out fold with it. Scoring $\mathcal{D}_{\text{tr}}$ with $f_{\text{beh}}$
  trained on all of $\mathcal{D}_{\text{tr}}$ is stacking leakage: it lets
  $g_\phi$ fit an optimistically confident score it never sees at inference.}
\State Compute $\mathbf{v}_j$ for $j \in \mathcal{D}_{\text{tr}}$ as above,
  using each session's out-of-fold $f_{\text{beh}}^{(k)}$ for
  $p_{\text{beh},j}, \sigma_{\text{beh},j}$

\State \textit{// Stage 2: CTGAN augmentation (training split only)}
\State Fit CTGAN$_1$ on $\{(\mathbf{v}_j, 1) : y_j{=}1,\, j \in D_{\text{tr}}\}$
  for 50 epochs
\State Fit CTGAN$_0$ on $\{(\mathbf{v}_j, 0) : y_j{=}0,\, j \in D_{\text{tr}}\}$
  for 50 epochs
\State $\mathcal{D}_{\text{syn}} \gets
  \text{CTGAN}_1.\text{sample}(N_{\text{synth}}/2) \cup
  \text{CTGAN}_0.\text{sample}(N_{\text{synth}}/2)$
\State Clip $\mathcal{D}_{\text{syn}}$ to $[0,1]^6$; remove NaN rows
\State $\mathcal{D}_{\text{aug}} \gets \mathcal{D}_{\text{tr}} \cup \mathcal{D}_{\text{syn}}$

\State \textit{// Stage 3: Decision MLP training}
\State Fit StandardScaler on $\mathcal{D}_{\text{aug}}$
\State Initialise MLP $g_\phi$ ($6{\to}32{\to}1$, Dropout $p{=}0.2$)
\State Train $g_\phi$ on $\mathcal{D}_{\text{aug}}$ with Adam $\eta{=}10^{-3}$,
  early stopping on $\mathcal{D}_{\text{val}}$ loss (patience $= 6$)
\State Restore best weights

\State \textit{// Stage 4: Temperature calibration (matches eq.~10 of the
  main paper; ECE is non-smooth and unsuitable as an L-BFGS objective)}
\State Let $\ell = \text{logit}(g_\phi(\mathbf{v}))$ (pre-sigmoid output)
\State $T^* \gets \argmin_{T>0} \text{NLL}_T(\mathcal{D}_{\text{val}})$ via
  L-BFGS, where $\text{NLL}_T = -\sum_j y_j \log \sigma(\ell_j/T) +
  (1{-}y_j)\log(1{-}\sigma(\ell_j/T))$
\State $\hat{p}_{\text{cal}} \gets \sigma(\ell/T^*)$

\State \textit{// Stage 5: Bayesian TPE policy optimisation}
\State Run $300$ Optuna TPE trials on $\mathcal{D}_{\text{cal}}$:
  $\Theta^* \gets \argmax_\Theta J(\Theta) = \text{bal\_acc}(\Theta)
  - \lambda \cdot \text{esc\_rate}(\Theta)$
\State \Return $g_\phi$, $T^*$, $\Theta^*$
\end{algorithmic}
\end{algorithm}

\section{Comparison with Hirano and Kobayashi}\label{sec:supp_hirano}
% ─────────────────────────────────────────────────────────────────────────────

The closest direct comparator is the RanSMAP detector of Hirano and Kobayashi, which
operates on the same hypervisor-level telemetry and is evaluated on the same
dataset.
Architectural differences limit strict metric comparisons: Hirano and Kobayashi\ use
a 1D-CNN+LSTM trained on 23D features ($\chi_s$ 5D storage $+$ $\chi_m$ 18D
memory) with a $0.1$-second window aggregated over a $30$-second detection
window and report F-score as their primary metric, while our Behavioural CNN
uses 16D features ($5$D storage $+$ $11$D memory) with a $10$-second window
and reports AUROC.

On the hardware-holdout partition---the closest common ground---Hirano and Kobayashi\
report F-score $0.856$, while our Behavioural CNN achieves AUROC $0.939$ on
the same split.
On the Conti sub-variant partition, Hirano and Kobayashi\ report mean recall $0.56$
on their full 23D feature set; our Behavioural CNN achieves mean recall
$0.68$, despite using seven fewer input dimensions and no LSTM temporal
integration.
This suggests that the $10$-second window captures sufficient temporal context
without recurrent processing, at substantially lower architectural complexity.

The primary qualitative advantage of the proposed framework is the uncertainty
signal.
Hirano and Kobayashi's output is a binary classification without any measure of model
confidence.
On the Conti sub-variants and concurrent-execution scenarios where both
frameworks struggle, the proposed system escalates rather than silently
misclassifies: the inverse relationship between per-variant recall and
escalation rate confirms that $\sigma_{\text{beh}}$ reliably identifies samples
most likely to be misclassified, transforming undetected attacks into escalated
events.

A second, more recent comparator is Ranker~\cite{zhang2024ranker}, a
kernel-level behavioural ransomware detector evaluated on 68 families,
reporting F1 $0.9943$ and false-positive rate $0.11\%$, with $97.16\%$
accuracy on 22 held-out families. These headline numbers exceed the AUROC
figures reported here, but the comparison is not like-for-like---different
dataset, different metric (F1/accuracy vs.\ AUROC), and no disclosed
protocol for confirming held-out families are genuinely distinct rather
than close variants of trained-on families, a distinction that matters
considerably: our own leave-one-family-out results
(Section~\ref{sec:supp_eaeg}) overturn a previously-reported perfect
zero-shot transfer once the family is honestly excluded. More
fundamentally, Ranker, like Hirano and Kobayashi's detector, produces a
single deterministic verdict, with no calibrated risk score and no
mechanism to route uncertain cases to analyst review---the uncertainty
quantification and formal coverage guarantees central to this paper
remain, to our knowledge, absent from the current kernel/behavioural
ransomware-detection literature.

\section{Adaptive Thresholding}\label{sec:supp_adaptive}
% ─────────────────────────────────────────────────────────────────────────────

The Bayesian TPE policy optimisation sets four thresholds
$(\tau_{\text{high}}^*, \tau^*, \kappa^*, \kappa_{\text{low}}^*)$ once on
the calibration split and holds them fixed for the lifetime of the deployment.
This is appropriate when the operational cost parameter $\lambda$ is stable
and the threat distribution is stationary.
In practice, neither assumption holds indefinitely: new ransomware families
shift the empirical risk-score distribution, and organisational risk tolerance
changes in response to operational conditions.

A principled extension would treat $(\tau, \kappa)$ as parameters of a
Bayesian posterior updated online from rolling analyst feedback.
Each analyst-confirmed or analyst-corrected triage decision provides a noisy
observation of whether the current threshold was appropriate; a
sliding-window estimate of $J(\tau, \kappa)$ on recent confirmed labels would
allow thresholds to recalibrate without retraining the underlying agents.
The Expert Feedback Loop already provides the infrastructure for incremental
Decision MLP updates; adaptive thresholding would operate at the policy layer
above it, requiring no change to agent or fusion model weights.
We defer this extension to future work.

\section{Detailed Computational Complexity}\label{sec:supp_complexity}
% ─────────────────────────────────────────────────────────────────────────────

Per-sample inference cost is dominated by MC Dropout: each agent performs $T$
independent forward passes, giving $\mathcal{O}(T \cdot P_i)$ operations per
agent for a fixed-dimensional input, where $P_i$ is the parameter count of
agent $i$.
Our implementation executes the three agents sequentially, so the measured
inference cost is $\mathcal{O}(T \cdot \sum_i P_i)$; the components could in
principle be parallelised across separate devices to bound latency by
$\mathcal{O}(T \cdot \max_i P_i)$ (the Semantic CNN, input
$\mathbb{R}^{384}$, would set that bound), but the latency figures reported
in the main paper are the measured sequential cost rather than a projected
parallel one.
The Decision MLP contributes $\mathcal{O}(1)$ (fixed $6{\to}32{\to}1$
architecture).
Behavioural feature extraction costs $\mathcal{O}(|E_w|)$ per window, where
$|E_w|$ is the number of hypervisor events in the $10$-second window ---
linear in telemetry volume and independent of model size.

Offline training costs are incurred once at deployment time.
CTGAN fitting scales as $\mathcal{O}(E_{\text{CTGAN}} \cdot N_{\text{tr}}
\cdot d_v)$ with $E_{\text{CTGAN}}{=}50$ epochs, $N_{\text{tr}}$ the
training set size, and $d_v{=}6$ the score-pool dimension.
Bayesian TPE optimisation is $\mathcal{O}(K_{\max}^{2} \log K_{\max})$
over $K_{\max}{=}300$ trials, as each trial's density estimation evaluates
the $\ell(x)$ and $g(x)$ Parzen estimators over all prior observations.
The Expert Feedback Loop incurs an online cost of
$\mathcal{O}(N_F \cdot |\phi|)$ per feedback iteration---computing the
diagonal Fisher over $N_F{=}1{,}000$ calibration samples and updating
$|\phi|$ MLP parameters---which completes in under one second on any
modern GPU.

\begin{figure}[h]

\textbf{Statistical significance.} We assess whether the Decision MLP's
AUROC exceeds the Behavioural CNN's with a one-sample Wilcoxon signed-rank
test (six per-seed Decision MLP AUROCs vs.\ the single, fixed, deployed
Behavioural CNN checkpoint, H$_1$: MLP $>$ B), since B is not retrained per
seed at deployment. It fails to reject the null on every split ($p{=}1.00$
throughout); the Behavioural CNN exceeds the Decision MLP by
$0.009$--$0.029$ AUROC on all four splits (e.g.\ random-stratified
$0.963$ vs.\ $0.952{\pm}0.0004$; family holdout $0.707{\pm}0.010$ vs.\
$0.677{\pm}0.001$). This is the expected consequence of restricting the
Semantic and Knowledge Agents to a genuinely label-independent corpus-mean
score for RanSMAP fusion (Remark 1 of the main paper): fusion cannot add
per-sample discriminative signal over the Behavioural Agent alone, so its
measured benefit is the uncertainty-gated triage policy it supports rather
than AUROC or calibration (Section~IV-D of the main paper, following the
RansKnow correction). A session-level McNemar test on random-stratified (majority-vote
aggregation, avoiding the within-session window correlation that
invalidates a window-level test) finds only $1$ disagreement out of $267$
sessions ($\chi^2{=}0$, $p{=}1.00$); the corresponding window-level test is
nominally significant ($\chi^2{=}54.4$, $p{<}10^{-4}$, $B$ correct/MLP
wrong on $257$ windows vs.\ $114$ the other way) but is not
independence-respecting.

\textbf{Selective precision-coverage.} Sweeping the auto-contain threshold
$\tau$ from $1.0$ to $0.0$ and tracing precision against auto-classified
coverage, the Area Under the Selective Precision--Coverage curve (AUSPC;
higher is better, not the conventional risk-coverage curve) is $0.829$
on random-stratified and $0.823$ on hardware holdout; temporal ($0.675$)
and family ($0.651$) holdout are lower, consistent with the same
moderate-vs-severe shift pattern seen in the main paper's AUROC table.

\textbf{Session-level detection and latency.} Aggregating Decision MLP
predictions to the session level: majority-vote (flag if $>50\%$ of
windows exceed $\hat{p}_{\text{MLP}}{\geq}0.5$) achieves near-perfect
in-distribution detection (random $100\%$, hardware $99.4\%$, temporal
$100\%$) at low session FPR ($3.0$--$6.1\%$), falling to $70.7\%$ detection
(FPR $19.7\%$) on family holdout; max-score (flag if any window exceeds
threshold) reaches $100\%$ detection on three of four splits but at much
higher FPR ($81$--$98\%$), confirming majority-vote is the more reliable
session-level rule under honestly out-of-fold scoring. Median latency to
first detection is $0\,\text{s}$ across all splits; mean latency rises
modestly under distribution shift (random $4.6\,\text{s}$ vs.\ family
holdout $6.1\,\text{s}$).

Two conclusions follow. First, the constant Knowledge dimension improves
calibration substantially (E-J1: ECE $0.040 \to 0.018$) at negligible
AUROC cost -- it functions as a learned prior the Decision MLP can shift
its output distribution against, not a discriminative signal. Second, the
deployed model's \emph{discrimination} is essentially insensitive to what
the Knowledge input actually is (E-J2: AUROC varies by $<0.001$ across

Peer Consultation (Definition~1 of the main paper) only substitutes an
agent's probability when \emph{that agent's own} uncertainty exceeds
$\kappa_{\text{PC}}$, so a deliberately low-uncertainty adversarial
Knowledge input is not caught by this safeguard, even though its practical
impact here is limited to calibration rather than detection.

\subsection{E-K: Cost-Sensitive Triage Objective}\label{sec:supp_cost}

Table~\ref{tab:cost_sensitive} reports the full confusion breakdown the
main paper's balanced-accuracy objective does not expose. We define
$J_{\text{cost}}(\Theta) = -(w_{\text{fn}} P_{\text{ransomware allowed}} +
w_{\text{fp}} P_{\text{benign auto-contained}} + w_{\text{esc}}
P_{\text{escalated}})$ with $w_{\text{fn}}{=}10$, $w_{\text{fp}}{=}5$,
$w_{\text{esc}}{=}1$ -- an illustrative weighting, not a claim of the
"correct" costs for every deployment, reflecting only that missing
ransomware is worse than a false auto-contain, which is worse than an
unnecessary escalation. Both policies are fit by the same Optuna TPE
procedure (150 trials) on a held-out validation split and evaluated on the
genuinely held-out random-stratified test split.

\section{Calibration-Set Separation for Temperature Scaling and TPE}\label{sec:supp_calset}
% ─────────────────────────────────────────────────────────────────────────────

The Decision MLP's temperature scaling and Bayesian TPE triage-policy
optimisation are fit on a stratified sample drawn independently of the
session-level train/val/test partitions used elsewhere in this paper,
rather than on a dedicated held-out slice of the same partition scheme.
We have not audited this sample for overlap with the evaluation splits'
training sessions. A rigorous re-fit on a clean, partition-consistent
calibration set---with temperature scaling, TPE, and conformal
calibration each using their own disjoint slice, as recommended in peer
review---is deferred to future work; all AUROC, calibration, and LOFO
numbers reported in the main paper are unaffected, since they do not
depend on the temperature or policy thresholds' fitting data.
