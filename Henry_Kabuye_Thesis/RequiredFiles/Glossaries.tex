\newglossaryentry{latex}{
    name=\LaTeX,
    description={A typesetting language used as a document preparation system,
    widely adopted for academic and scientific publications.}
}

\newglossaryentry{epistemic}{
    name={epistemic uncertainty},
    description={Uncertainty arising from a model's lack of knowledge, typically
    caused by sparse training data or out-of-distribution inputs. Unlike aleatoric
    uncertainty, epistemic uncertainty is reducible by collecting more data.}
}

\newglossaryentry{aleatoric}{
    name={aleatoric uncertainty},
    description={Irreducible uncertainty inherent in the data itself, arising from
    noise, measurement limitations, or genuine ambiguity at the decision boundary.
    Cannot be reduced by collecting more training data.}
}

\newglossaryentry{calibration}{
    name={probability calibration},
    description={The process of adjusting a classifier's posterior probability
    estimates so that a predicted probability of $p$ corresponds to an empirical
    correctness frequency of $p$ across a population of predictions.}
}

\newglossaryentry{triage}{
    name={triage policy},
    description={A decision rule that routes model predictions to different response
    pathways — autonomous action, human escalation, or allow-through — based on
    predicted probability and uncertainty thresholds.}
}

\newglossaryentry{conceptdrift}{
    name={concept drift},
    description={A change over time in the statistical relationship between input
    features and class labels, causing a classifier trained on historical data to
    degrade in performance on more recently collected samples.}
}

\newglossaryentry{experiencereplay}{
    name={experience replay},
    description={A continual learning technique in which a buffer of previously
    observed labelled examples is maintained and sampled alongside new examples
    during training, mitigating catastrophic forgetting.}
}

\newglossaryentry{counterfactual}{
    name={counterfactual explanation},
    description={An explanation for a model prediction that identifies the minimal
    perturbation of the input feature vector that would reverse the predicted class,
    providing actionable guidance to analysts on which features drove the decision.}
}

\newglossaryentry{neurosymbolic}{
    name={neuro-symbolic AI},
    description={A class of AI architectures that integrate neural learning components,
    capable of acquiring representations from data, with symbolic reasoning components
    that encode structured knowledge, logical constraints, or semantic relationships.}
}
